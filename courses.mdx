import numpy as np

def sigmoid(x):
    return 1 / (1 + np.exp(-x))

class SimpleNN:
    def __init__(self):
        self.weights = np.random.rand(2, 1)
    
    def forward(self, x):
        return sigmoid(np.dot(x, self.weights))
